create view public.vw_rfid_loyalty_cards as
select
  card.id as card_id,
  card.uid,
  card.status as card_status,
  card.customer_id,
  case
    when c.id is not null then COALESCE(
      NULLIF(
        TRIM(
          both
          from
            (c.first_name::text || ' '::text) || c.last_name::text
        ),
        ''::text
      ),
      c.customer_code::text,
      'Customer #'::text || c.id
    )
    else 'Unassigned'::text
  end as owner_name,
  c.phone as owner_phone,
  c.email as owner_email,
  c.customer_type,
  c.status as customer_status,
  c.customer_code,
  card.total_orders,
  card.total_loads,
  card.points_balance,
  card.created_at as card_created_at,
  card.updated_at as card_updated_at,
  p.username as account_username,
  p.email as account_email,
  p.role as account_role
from
  rfid_loyalty_cards card
  left join customers c on card.customer_id = c.id
  left join profiles p on c.profile_id = p.id;
